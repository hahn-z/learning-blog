---# PHP结构型设计模式

结构型模式关注类和对象的组合方式，解决接口不兼容、功能扩展、结构简化等问题。PHP 常用的结构型模式有：适配器、桥接、组合、装饰器、外观、享元、代理。

## 适配器模式（Adapter）

将一个类的接口转换成客户端期望的另一个接口，使不兼容的类可以协同工作。

```php
<?php
interface NotificationInterface
{
    public function send(string $recipient, string $message): bool;
}

// 第三方 SDK（不兼容的接口）
class ThirdPartySmsSdk
{
    public function dispatchSms(string $phone, string $text, array $options = []): int
    {
        echo "SMS to $phone: $text\n";
        return 200;
    }
}

// 适配器
class SmsAdapter implements NotificationInterface
{
    public function __construct(
        private ThirdPartySmsSdk $sdk,
        private array $defaultOptions = [],
    ) {}

    public function send(string $recipient, string $message): bool
    {
        $result = $this->sdk->dispatchSms($recipient, $message, $this->defaultOptions);
        return $result === 200;
    }
}

// 使用统一的接口
function notifyUser(NotificationInterface $notifier, string $to, string $msg): void
{
    $notifier->send($to, $msg);
}

notifyUser(new SmsAdapter(new ThirdPartySmsSdk()), '13800138000', 'Hello');
```

## 桥接模式（Bridge）

将抽象部分与实现部分分离，使两者可以独立变化。

```php
<?php
// 实现接口
interface Renderer
{
    public function renderTitle(string $title): string;
    public function renderText(string $text): string;
}

class HtmlRenderer implements Renderer
{
    public function renderTitle(string $title): string { return "<h1>$title</h1>"; }
    public function renderText(string $text): string { return "<p>$text</p>"; }
}

class MarkdownRenderer implements Renderer
{
    public function renderTitle(string $title): string { return "# $title\n"; }
    public function renderText(string $text): string { return "$text\n"; }
}

// 抽象部分
abstract class Page
{
    public function __construct(
        protected Renderer $renderer,
    ) {}

    abstract public function view(): string;
}

class SimplePage extends Page
{
    public function __construct(
        Renderer $renderer,
        private string $title,
        private string $content,
    ) {
        parent::__construct($renderer);
    }

    public function view(): string
    {
        return $this->renderer->renderTitle($this->title)
             . $this->renderer->renderText($this->content);
    }
}

// 桥接：同一页面可以用不同渲染器
$htmlPage = new SimplePage(new HtmlRenderer(), 'Hello', 'World');
$mdPage   = new SimplePage(new MarkdownRenderer(), 'Hello', 'World');

echo $htmlPage->view(); // <h1>Hello</h1><p>World</p>
echo $mdPage->view();   // # Hello\nWorld\n
```

## 组合模式（Composite）

将对象组合成树形结构，统一处理单个对象和组合对象。

```php
<?php
interface Renderable
{
    public function render(): string;
}

class Field implements Renderable
{
    public function __construct(
        private string $label,
        private string $name,
        private string $type = 'text',
    ) {}

    public function render(): string
    {
        return "<label>$label</label><input type=\"$type\" name=\"$name\" />";
    }
}

class FieldGroup implements Renderable
{
    /** @var Renderable[] */
    private array $children = [];

    public function add(Renderable $element): void
    {
        $this->children[] = $element;
    }

    public function render(): string
    {
        $html = '<fieldset>';
        foreach ($this->children as $child) {
            $html .= $child->render();
        }
        return $html . '</fieldset>';
    }
}

// 树形结构：组中可以包含字段或子组
$personalGroup = new FieldGroup();
$personalGroup->add(new Field('Name', 'name'));
$personalGroup->add(new Field('Email', 'email', 'email'));

$addressGroup = new FieldGroup();
$addressGroup->add(new Field('City', 'city'));
$addressGroup->add(new Field('Zip', 'zip'));

$form = new FieldGroup();
$form->add($personalGroup);
$form->add($addressGroup);

echo $form->render();
```

## 装饰器模式（Decorator）

动态地给对象添加职责，比继承更灵活。

```php
<?php
interface Message
{
    public function getContent(): string;
}

class SimpleMessage implements Message
{
    public function __construct(private string $content) {}

    public function getContent(): string
    {
        return $this->content;
    }
}

abstract class MessageDecorator implements Message
{
    public function __construct(protected Message $message) {}
}

class HtmlDecorator extends MessageDecorator
{
    public function getContent(): string
    {
        return '<p>' . $this->message->getContent() . '</p>';
    }
}

class UpperCaseDecorator extends MessageDecorator
{
    public function getContent(): string
    {
        return strtoupper($this->message->getContent());
    }
}

class TimestampDecorator extends MessageDecorator
{
    public function getContent(): string
    {
        return '[' . date('Y-m-d H:i') . '] ' . $this->message->getContent();
    }
}

// 自由组合装饰器
$msg = new SimpleMessage('Hello World');
$msg = new TimestampDecorator(new UpperCaseDecorator(new HtmlDecorator($msg)));
echo $msg->getContent();
// <p>[2026-05-31 10:00] HELLO WORLD</p>
```

## 外观模式（Facade）

为复杂子系统提供简化的统一接口。

```php
<?php
class VideoConverter
{
    public function convert(string $file, string $format): string
    {
        // 复杂子系统：编解码器选择、音频提取、格式转换、压缩
        $codec = (new CodecFactory())->create($format);
        $audio = (new AudioExtractor())->extract($file);
        $video = (new VideoProcessor())->process($file, $codec);
        return (new Compressor())->compress($video, $audio, $codec);
    }
}

// 客户端只需调用一个方法
$converter = new VideoConverter();
$result = $converter->convert('movie.avi', 'mp4');
```

**Laravel 的 Facade：** Laravel 的 Facade 不是严格的外观模式，而是静态代理——通过 `__callStatic` 转发调用到 IoC 容器中的实际服务。

## 享元模式（Flyweight）

共享对象以减少内存使用，适合大量相似对象的场景。

```php
<?php
class CharacterStyle
{
    public function __construct(
        private string $font,
        private int $size,
        private string $color,
    ) {}

    public function getFont(): string { return $this->font; }
    public function getSize(): int { return $this->size; }
    public function getColor(): string { return $this->color; }
}

// 享元工厂 — 共享相同属性的对象
class StyleFactory
{
    private static array $pool = [];

    public static function getStyle(string $font, int $size, string $color): CharacterStyle
    {
        $key = "$font:$size:$color";
        return self::$pool[$key] ??= new CharacterStyle($font, $size, $color);
    }

    public static function getPoolSize(): int
    {
        return count(self::$pool);
    }
}

// 10万个字符，但只有几种样式
$text = str_repeat('Hello World ', 10000);
$styles = [
    StyleFactory::getStyle('Arial', 12, 'black'),
    StyleFactory::getStyle('Arial', 14, 'red'),
    StyleFactory::getStyle('Times', 12, 'black'),
];

echo StyleFactory::getPoolSize(); // 3 — 只有3个样式对象
```

## 代理模式（Proxy）

控制对另一个对象的访问，添加间接层。

```php
<?php
interface Image
{
    public function display(): string;
}

class RealImage implements Image
{
    public function __construct(private string $filename)
    {
        // 模拟昂贵的加载操作
        echo "Loading $filename from disk...\n";
    }

    public function display(): string
    {
        return "Displaying $filename";
    }
}

class LazyImageProxy implements Image
{
    private ?RealImage $realImage = null;

    public function __construct(private string $filename) {}

    public function display(): string
    {
        // 延迟加载 — 只在第一次 display 时加载
        $this->realImage ??= new RealImage($this->filename);
        return $this->realImage->display();
    }
}

$proxy = new LazyImageProxy('large-photo.png'); // 不立即加载
echo $proxy->display(); // 此时才加载
```

## 面试常见追问

**Q: 适配器和装饰器的区别？**

A: 适配器改变接口（让不兼容的接口协同工作），装饰器增强功能（不改变接口，添加职责）。适配器是事后补救，装饰器是预先设计。

**Q: 装饰器和继承的区别？**

A: 继承是静态的（编译时确定），装饰器是动态的（运行时组合）。继承导致类爆炸（N个功能需要 2^N 个子类），装饰器只需 N 个装饰器类自由组合。

**Q: 代理和适配器的区别？**

A: 代理提供相同的接口（控制访问），适配器提供不同的接口（接口转换）。代理可以添加缓存、权限控制、延迟加载等横切关注点。

## 小结

| 模式 | 核心意图 | 典型场景 |
|------|---------|---------|
| 适配器 | 接口转换 | 第三方库集成 |
| 桥接 | 抽象与实现分离 | 多维度变化 |
| 组合 | 树形结构统一处理 | UI组件、文件系统 |
| 装饰器 | 动态添加职责 | 中间件、日志、缓存 |
| 外观 | 简化复杂接口 | 子系统封装 |
| 享元 | 共享减少内存 | 大量相似对象 |
| 代理 | 控制对象访问 | 延迟加载、权限控制 |
