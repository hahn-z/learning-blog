---# PHP行为型设计模式

行为型模式关注对象之间的通信和职责分配。PHP 常用的行为型模式有：观察者、策略、命令、责任链、状态、模板方法、迭代器。

## 观察者模式（Observer）

定义一对多的依赖关系，当一个对象状态改变时自动通知所有依赖者。

```php
<?php
// PHP 内置 SplSubject / SplObserver
use SplSubject, SplObserver, SplObjectStorage;

class EventBus implements SplSubject
{
    private SplObjectStorage $observers;

    public function __construct(
        private string $name = '',
        private mixed $data = null,
    ) {
        $this->observers = new SplObjectStorage();
    }

    public function attach(SplObserver $observer): void
    {
        $this->observers->attach($observer);
    }

    public function detach(SplObserver $observer): void
    {
        $this->observers->detach($observer);
    }

    public function notify(): void
    {
        foreach ($this->observers as $observer) {
            $observer->update($this);
        }
    }

    public function fire(string $name, mixed $data = null): void
    {
        $this->name = $name;
        $this->data = $data;
        $this->notify();
    }

    public function getName(): string { return $this->name; }
    public function getData(): mixed { return $this->data; }
}

class LoggerSubscriber implements SplObserver
{
    public function update(SplSubject $subject): void
    {
        echo "[LOG] Event: {$subject->getName()}\n";
    }
}

class EmailSubscriber implements SplObserver
{
    public function update(SplSubject $subject): void
    {
        echo "[EMAIL] Notification for: {$subject->getName()}\n";
    }
}

$bus = new EventBus();
$bus->attach(new LoggerSubscriber());
$bus->attach(new EmailSubscriber());

$bus->fire('user.registered', ['id' => 1]);
// [LOG] Event: user.registered
// [EMAIL] Notification for: user.registered
```

## 策略模式（Strategy）

定义一系列算法，将每个算法封装起来，使它们可以互换。

```php
<?php
interface ShippingStrategy
{
    public function calculate(float $weight, string $destination): float;
}

class StandardShipping implements ShippingStrategy
{
    public function calculate(float $weight, string $destination): float
    {
        return $weight * 5.0 + ($destination === 'remote' ? 20 : 0);
    }
}

class ExpressShipping implements ShippingStrategy
{
    public function calculate(float $weight, string $destination): float
    {
        return $weight * 10.0 + ($destination === 'remote' ? 50 : 10);
    }
}

class FreeShipping implements ShippingStrategy
{
    public function calculate(float $weight, string $destination): float
    {
        return 0;
    }
}

class ShippingCalculator
{
    public function __construct(private ShippingStrategy $strategy) {}

    public function setStrategy(ShippingStrategy $strategy): void
    {
        $this->strategy = $strategy;
    }

    public function calculate(float $weight, string $destination): float
    {
        return $this->strategy->calculate($weight, $destination);
    }
}

$calc = new ShippingCalculator(new StandardShipping());
echo $calc->calculate(5, 'normal'); // 25
$calc->setStrategy(new ExpressShipping());
echo $calc->calculate(5, 'normal'); // 60
```

## 命令模式（Command）

将请求封装为对象，支持撤销、队列、日志等操作。

```php
<?php
interface Command
{
    public function execute(): void;
    public function undo(): void;
}

class Light
{
    public function on(): void  { echo "Light ON\n"; }
    public function off(): void { echo "Light OFF\n"; }
}

class LightOnCommand implements Command
{
    public function __construct(private Light $light) {}

    public function execute(): void { $this->light->on(); }
    public function undo(): void    { $this->light->off(); }
}

class LightOffCommand implements Command
{
    public function __construct(private Light $light) {}

    public function execute(): void { $this->light->off(); }
    public function undo(): void    { $this->light->on(); }
}

class RemoteControl
{
    private array $history = [];

    public function submit(Command $command): void
    {
        $command->execute();
        $this->history[] = $command;
    }

    public function undoLast(): void
    {
        if (empty($this->history)) return;
        $command = array_pop($this->history);
        $command->undo();
    }
}

$light = new Light();
$remote = new RemoteControl();

$remote->submit(new LightOnCommand($light));  // Light ON
$remote->submit(new LightOffCommand($light)); // Light OFF
$remote->undoLast();                          // Light ON（撤销关闭）
$remote->undoLast();                          // Light OFF（撤销打开）
```

## 责任链模式（Chain of Responsibility）

将请求沿处理链传递，每个处理者决定是否处理或传递给下一个。

```php
<?php
abstract class Handler
{
    private ?Handler $next = null;

    public function setNext(Handler $handler): Handler
    {
        $this->next = $handler;
        return $handler;
    }

    public function handle(Request $request): ?string
    {
        if ($this->next) {
            return $this->next->handle($request);
        }
        return null;
    }
}

class Request
{
    public function __construct(
        public string $type,
        public array $data = [],
    ) {}
}

class AuthHandler extends Handler
{
    public function handle(Request $request): ?string
    {
        if (empty($request->data['token'])) {
            return 'Unauthorized';
        }
        return parent::handle($request);
    }
}

class RateLimitHandler extends Handler
{
    public function handle(Request $request): ?string
    {
        if (($request->data['requests'] ?? 0) > 100) {
            return 'Rate Limited';
        }
        return parent::handle($request);
    }
}

class BusinessHandler extends Handler
{
    public function handle(Request $request): ?string
    {
        return "Processed: {$request->type}";
    }
}

// 构建链
$auth = new AuthHandler();
$auth->setNext(new RateLimitHandler())
     ->setNext(new BusinessHandler());

echo $auth->handle(new Request('api', ['token' => 'abc', 'requests' => 5]));
// Processed: api

echo $auth->handle(new Request('api', [])); // Unauthorized
```

## 状态模式（State）

允许对象在内部状态改变时改变其行为。

```php
<?php
interface OrderState
{
    public function pay(OrderContext $order): void;
    public function ship(OrderContext $order): void;
    public function deliver(OrderContext $order): void;
    public function cancel(OrderContext $order): void;
}

class PendingState implements OrderState
{
    public function pay(OrderContext $order): void
    {
        echo "Order paid\n";
        $order->setState(new PaidState());
    }
    public function ship(OrderContext $order): void { echo "Cannot ship unpaid order\n"; }
    public function deliver(OrderContext $order): void { echo "Cannot deliver unpaid order\n"; }
    public function cancel(OrderContext $order): void
    {
        echo "Order cancelled\n";
        $order->setState(new CancelledState());
    }
}

class PaidState implements OrderState
{
    public function pay(OrderContext $order): void { echo "Already paid\n"; }
    public function ship(OrderContext $order): void
    {
        echo "Order shipped\n";
        $order->setState(new ShippedState());
    }
    public function deliver(OrderContext $order): void { echo "Cannot deliver unshipped order\n"; }
    public function cancel(OrderContext $order): void
    {
        echo "Refunding and cancelling\n";
        $order->setState(new CancelledState());
    }
}

class ShippedState implements OrderState
{
    public function pay(OrderContext $order): void { echo "Already paid\n"; }
    public function ship(OrderContext $order): void { echo "Already shipped\n"; }
    public function deliver(OrderContext $order): void
    {
        echo "Order delivered\n";
        $order->setState(new DeliveredState());
    }
    public function cancel(OrderContext $order): void { echo "Cannot cancel shipped order\n"; }
}

// 简化：DeliveredState 和 CancelledState 类似
class DeliveredState implements OrderState
{
    public function pay(OrderContext $o): void { echo "Completed\n"; }
    public function ship(OrderContext $o): void { echo "Completed\n"; }
    public function deliver(OrderContext $o): void { echo "Already delivered\n"; }
    public function cancel(OrderContext $o): void { echo "Completed\n"; }
}
class CancelledState implements OrderState
{
    public function pay(OrderContext $o): void { echo "Cancelled\n"; }
    public function ship(OrderContext $o): void { echo "Cancelled\n"; }
    public function deliver(OrderContext $o): void { echo "Cancelled\n"; }
    public function cancel(OrderContext $o): void { echo "Already cancelled\n"; }
}

class OrderContext
{
    public function __construct(
        private OrderState $state = new PendingState(),
    ) {}

    public function setState(OrderState $state): void { $this->state = $state; }
    public function pay(): void     { $this->state->pay($this); }
    public function ship(): void    { $this->state->ship($this); }
    public function deliver(): void { $this->state->deliver($this); }
    public function cancel(): void  { $this->state->cancel($this); }
}

$order = new OrderContext();
$order->pay();     // Order paid
$order->ship();    // Order shipped
$order->deliver(); // Order delivered
```

## 模板方法模式（Template Method）

在父类定义算法骨架，子类实现具体步骤。

```php
<?php
abstract class DataImporter
{
    // 模板方法 — 定义算法骨架（final 防止覆盖）
    final public function import(string $source): void
    {
        $data = $this->read($source);
        $validated = $this->validate($data);
        $transformed = $this->transform($validated);
        $this->save($transformed);
        $this->notify();
    }

    abstract protected function read(string $source): array;
    abstract protected function transform(array $data): array;

    protected function validate(array $data): array
    {
        return array_filter($data, fn($row) => !empty($row));
    }

    protected function save(array $data): void
    {
        echo "Saving " . count($data) . " records\n";
    }

    protected function notify(): void
    {
        echo "Import complete\n";
    }
}

class CsvImporter extends DataImporter
{
    protected function read(string $source): array
    {
        $rows = array_map('str_getcsv', file($source));
        return array_slice($rows, 1); // 跳过表头
    }

    protected function transform(array $data): array
    {
        return array_map(fn($row) => ['name' => $row[0], 'email' => $row[1]], $data);
    }
}

class ApiImporter extends DataImporter
{
    protected function read(string $source): array
    {
        return json_decode(file_get_contents($source), true);
    }

    protected function transform(array $data): array
    {
        return array_map(fn($row) => [
            'name'  => $row['full_name'],
            'email' => $row['email_address'],
        ], $data);
    }
}
```

## 面试常见追问

**Q: 策略模式和状态模式的区别？**

A: 策略模式由客户端选择策略（外部切换），状态模式由对象自己根据状态切换行为（内部变化）。策略强调"可替换的算法"，状态强调"状态驱动的行为"。

**Q: 观察者模式有什么缺点？**

A: (1) 隐式调用——不容易追踪哪些观察者被通知；(2) 内存泄漏——忘记取消注册的观察者不会被 GC；(3) 级联更新——观察者修改被观察者状态可能导致无限循环。

**Q: 责任链和装饰器的区别？**

A: 结构相似（都是链式传递），但意图不同。装饰器每层都处理并增强（管道），责任链在某一层决定是否处理或传递（拦截）。

**Q: 模板方法为什么要用 `final`？**

A: 防止子类改变算法骨架。模板方法的核心价值就是固定流程、允许扩展步骤。去掉 `final` 就失去了这个保证。

## 小结

| 模式 | 核心意图 | 典型场景 |
|------|---------|---------|
| 观察者 | 一对多通知 | 事件系统、消息订阅 |
| 策略 | 算法互换 | 支付方式、排序策略 |
| 命令 | 请求对象化 | 撤销/重做、任务队列 |
| 责任链 | 逐级处理 | 中间件、审批流 |
| 状态 | 状态驱动行为 | 订单状态机、工作流 |
| 模板方法 | 固定骨架、扩展步骤 | 框架钩子、数据导入 |
